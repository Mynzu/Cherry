
CREATE TABLE estados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL
);

CREATE TABLE municipios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    estado_id INT NOT NULL,
    FOREIGN KEY (estado_id) REFERENCES estados(id) ON DELETE CASCADE
);

CREATE TABLE parroquias (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    municipio_id INT NOT NULL,
    FOREIGN KEY (municipio_id) REFERENCES municipios(id) ON DELETE CASCADE
);

CREATE TABLE subregiones (
    id INT AUTO_INCREMENT PRIMARY KEY,
    codigo_region VARCHAR(50) UNIQUE NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    estado_id INT NOT NULL,
    FOREIGN KEY (estado_id) REFERENCES estados(id) ON DELETE CASCADE
);

CREATE TABLE centros_salud (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_centro VARCHAR(50) UNIQUE NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    parroquia_id INT NOT NULL,
    FOREIGN KEY (parroquia_id) REFERENCES parroquias(id) ON DELETE CASCADE
);

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    rol ENUM('master', 'subregion', 'centro') NOT NULL,
    subregion_id INT NULL,
    centro_salud_id INT NULL,
    FOREIGN KEY (subregion_id) REFERENCES subregiones(id) ON DELETE SET NULL,
    FOREIGN KEY (centro_salud_id) REFERENCES centros_salud(id) ON DELETE SET NULL
);

CREATE TABLE personas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    primer_nombre VARCHAR(50) NOT NULL,
    segundo_nombre VARCHAR(50),
    primer_apellido VARCHAR(50) NOT NULL,
    segundo_apellido VARCHAR(50),
    cedula VARCHAR(20) UNIQUE NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    genero ENUM('M', 'F', 'O') NOT NULL,
    parroquia_id INT,
    FOREIGN KEY (parroquia_id) REFERENCES parroquias(id) ON DELETE SET NULL
);

CREATE TABLE vacunas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    dosis_requeridas INT DEFAULT 1
);

CREATE TABLE dosis_aplicadas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    persona_id INT NOT NULL,
    vacuna_id INT NOT NULL,
    dosis VARCHAR(50) NOT NULL, 
    fecha DATE NOT NULL,
    lote VARCHAR(50),
    centro_salud_id INT,
    usuario_registra INT, 
    observaciones TEXT,
    FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE,
    FOREIGN KEY (vacuna_id) REFERENCES vacunas(id) ON DELETE CASCADE,
    FOREIGN KEY (centro_salud_id) REFERENCES centros_salud(id) ON DELETE SET NULL,
    FOREIGN KEY (usuario_registra) REFERENCES usuarios(id) ON DELETE SET NULL
);

CREATE TABLE examenes_salud (
    id INT AUTO_INCREMENT PRIMARY KEY,
    persona_id INT NOT NULL,
    tipo_examen VARCHAR(100) NOT NULL,
    estatus ENUM('apto', 'no_apto') NOT NULL,
    especialista VARCHAR(150) NOT NULL,
    observaciones TEXT,
    usuario_registra INT,
    FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_registra) REFERENCES usuarios(id) ON DELETE SET NULL
);

CREATE TABLE revisiones_examenes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    examen_id INT NOT NULL,
    usuario_revisa_id INT NOT NULL,
    comentario_revision TEXT,
    fecha_revision DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (examen_id) REFERENCES examenes_salud(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_revisa_id) REFERENCES usuarios(id) ON DELETE CASCADE
);

CREATE TABLE certificados_salud (
    id INT AUTO_INCREMENT PRIMARY KEY,
    persona_id INT NOT NULL,
    usuario_emite_id INT NOT NULL,
    fecha_emision DATETIME DEFAULT CURRENT_TIMESTAMP,
    estado_certificado VARCHAR(50) DEFAULT 'Aprobado',
    FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_emite_id) REFERENCES usuarios(id) ON DELETE CASCADE
);

CREATE TABLE bitacora (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    usuario_nombre VARCHAR(100),
    accion TEXT NOT NULL,
    fecha DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL
);
